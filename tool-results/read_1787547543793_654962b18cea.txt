   103→   103→  const [data, setData] = useState<GuestHouse[]>([]);
   104→   104→  const [loading, setLoading] = useState(true);
   105→   105→  const [search, setSearch] = useState('');
   106→   106→  const [filterSubCity, setFilterSubCity] = useState('');
   107→   107→  const [filterArea, setFilterArea] = useState('');
   108→   108→  const [areas, setAreas] = useState<string[]>([]);
   109→   109→  const [page, setPage] = useState(1);
   110→   110→  const [totalPages, setTotalPages] = useState(1);
   111→   111→  const [total, setTotal] = useState(0);
   112→   112→  const [deleteId, setDeleteId] = useState<string | null>(null);
   113→   113→  const [viewItem, setViewItem] = useState<GuestHouse | null>(null);

   145→   145→  }, [search, filterSubCity, filterArea, page, toast]);

   254→   254→        title: 'Export Failed',

   272→   272→      a.href = url;
   273→   273→      a.download = `bishoftu_guesthouses_survey_${new Date().toISOString().split('T')[0]}.xlsx`;

   363→   363→      {/* Data Table */}

   380→   380→            </p>
   381→   381→          </CardContent>
   382→   382→        </Card>
   383→   383→      ) : (

   386→   386→            {/* Desktop Table */}
   387→   387→            <div className="hidden md:block overflow-x-auto">

   389→   389→                <thead>
