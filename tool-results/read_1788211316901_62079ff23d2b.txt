    21→import {

    31→import {
    32→  Dialog,

    35→  DialogHeader,
    36→  DialogTitle,
    37→  DialogFooter,
    38→} from '@/components/ui/dialog';
    39→import { Badge } from '@/components/ui/badge';
    40→import { Skeleton } from '@/components/ui/skeleton';
    41→import { useToast } from '@/hooks/use-toast';
    42→import {
    43→  Search,

    45→  Trash2,
    46→  Eye,
    47→  Building2,
    48→  ChevronLeft,

    56→  Plus,

    58→} from 'lucide-react';

    60→

    63→  guestHouseName: string;

    65→  subCity: string;
    66→  area: string;
    67→  specificAddress: string;
    68→  numberOfRooms: number;
    69→  licenseType: string;
    70→  licenseLevel: string;
    71→  licenseNumber: string | null;
    72→  serviceRating: number;
    73→  contactPhone: string | null;

    76→  hasRestaurant: boolean;
    77→  hasParking: boolean;
    78→  hasWiFi: boolean;
    79→  hasHotWater: boolean;
    80→  additionalServices: string | null;
    81→  surveyorName: string | null;
    82→  createdAt: string;
    83→  updatedAt: string;

    85→
    86→interface EditForm {
    87→  organizationName: string;
    88→  subCity: string;
    89→  area: string;
    90→  specificAddress: string;
    91→  numberOfRooms: string;
    92→  licenseType: string;
    93→  licenseLevel: string;
    94→  licenseNumber: string;
    95→  ownerName: string;
    96→  contactName: string;
    97→  contactPhone: string;
    98→}
    99→
   100→interface DataListProps {
   101→  refreshTrigger: number;
   102→}
   103→
   104→export default function DataList({ refreshTrigger }: DataListProps) {
   105→  const { toast } = useToast();
   106→  const [data, setData] = useState<GuestHouse[]>([]);
   107→  const [loading, setLoading] = useState(true);
   108→  const [search, setSearch] = useState('');
   109→  const [filterSubCity, setFilterSubCity] = useState('');
   110→  const [filterArea, setFilterArea] = useState('');
   111→  const [areas, setAreas] = useState<string[]>([]);
   112→  const [page, setPage] = useState(1);
   113→  const [totalPages, setTotalPages] = useState(1);
   114→  const [total, setTotal] = useState(0);
   115→  const [deleteId, setDeleteId] = useState<string | null>(null);
   116→  const [viewItem, setViewItem] = useState<GuestHouse | null>(null);
   117→  const [editItem, setEditItem] = useState<GuestHouse | null>(null);
   118→  const [editForm, setEditForm] = useState<EditForm | null>(null);
   119→  const [editAreas, setEditAreas] = useState<string[]>([]);
   120→  const [saving, setSaving] = useState(false);
   121→  const [deleting, setDeleting] = useState(false);
   122→  const [showRegister, setShowRegister] = useState(false);
   123→  const [showImport, setShowImport] = useState(false);
   124→  const [registering, setRegistering] = useState(false);
   125→  const [importing, setImporting] = useState(false);
   126→  const [importResult, setImportResult] = useState<{success: number, failed: number, errors: string[]} | null>(null);
   127→  const fileInputRef = useRef<HTMLInputElement>(null);
   128→  const [regSubCity, setRegSubCity] = useState('');
   129→  const [regAreas, setRegAreas] = useState<string[]>([]);
   130→  const [regForm, setRegForm] = useState({
   131→    organizationName: '', subCity: '', area: '', specificAddress: '',
   132→    numberOfRooms: '', licenseType: '', licenseLevel: '', licenseNumber: '',
   133→    ownerName: '', contactName: '', contactPhone: ''
   134→  });
   135→
   136→  const fetchData = useCallback(async () => {
   137→    setLoading(true);
   138→    try {
   139→      const params = new URLSearchParams();
   140→      if (search) params.set('search', search);
   141→      if (filterSubCity) params.set('subCity', filterSubCity);
   142→      if (filterArea) params.set('area', filterArea);
   143→      params.set('page', String(page));
   144→      params.set('limit', '50');
   145→
   146→      const res = await fetch(`/api/guesthouses?${params}`);
   147→      if (!res.ok) throw new Error('Failed to fetch');
   148→      const json = await res.json();
   149→      setData(json.data);
   150→      setTotal(json.total);
   151→      setTotalPages(json.totalPages);
   152→    } catch {
   153→      toast({
   154→        title: 'Error',
   155→        description: 'Failed to load data',
   156→        variant: 'destructive',
   157→      });
   158→    } finally {
   159→      setLoading(false);
   160→    }
   161→  }, [search, filterSubCity, filterArea, page, toast]);
   162→
   163→  useEffect(() => {
   164→    fetchData();
   165→  }, [fetchData, refreshTrigger]);
   166→
   167→  const handleSubCityFilterChange = (value: string) => {
   168→    setFilterSubCity(value);
   169→    setFilterArea('');
   170→    setPage(1);
   171→    if (value) {
   172→      const sc = locationData.subCities.find((s) => s.name === value);
   173→      setAreas(sc ? sc.areas : []);
   174→    } else {
   175→      setAreas([]);
   176→    }
   177→  };
   178→
   179→  const handleDelete = async () => {
   180→    if (!deleteId) return;
   181→    setDeleting(true);
   182→    try {
   183→      const res = await fetch(`/api/guesthouses/${deleteId}`, {
   184→        method: 'DELETE',
   185→      });
   186→      if (!res.ok) throw new Error('Delete failed');
   187→      toast({ title: 'Deleted', description: 'Record removed successfully.' });
   188→      fetchData();
   189→    } catch {
   190→      toast({
   191→        title: 'Delete Failed',
   192→        description: 'Could not delete the record.',
   193→        variant: 'destructive',
   194→      });
   195→    } finally {
   196→      setDeleting(false);
   197→      setDeleteId(null);
   198→    }
   199→  };
   200→
   201→  const handleRegister = async () => {
   202→    if (!regForm.organizationName || !regForm.subCity || !regForm.area || !regForm.specificAddress ||
   203→        !regForm.numberOfRooms || !regForm.licenseType || !regForm.licenseLevel || !regForm.licenseNumber ||
   204→        !regForm.ownerName || !regForm.contactName || !regForm.contactPhone) {
   205→      toast({ title: 'Qabiyyee Hin Jiru / Missing Fields', description: 'All fields are required', variant: 'destructive' });
   206→      return;
   207→    }
   208→    const phoneClean = regForm.contactPhone.replace(/[\s-]/g, '');
   209→    if (!/^(\+251|251|0)?(9|7)\d{8}$/.test(phoneClean)) {
   210→      toast({ title: 'Bilbila Dogoggora / Invalid Phone', description: 'Enter valid Ethiopian phone number', variant: 'destructive' });
   211→      return;
   212→    }
   213→    setRegistering(true);
   214→    try {
   215→      const res = await fetch('/api/guesthouses', {
   216→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({ ...regForm, serviceRating: 0, surveyorName: 'Admin' }),
   218→      });
   219→      if (!res.ok) { const err = await res.json(); throw new Error(err.error); }
   220→      toast({ title: 'Galmeen Galmeerrame! / Registered!', description: regForm.organizationName });
   221→      setShowRegister(false);
   222→      setRegForm({ organizationName: '', subCity: '', area: '', specificAddress: '', numberOfRooms: '', licenseType: '', licenseLevel: '', licenseNumber: '', ownerName: '', contactName: '', contactPhone: '' });
   223→      setRegSubCity(''); setRegAreas([]);
   224→      fetchData();
   225→    } catch (error) {
   226→      toast({ title: 'Hin Dhufne / Failed', description: error instanceof Error ? error.message : 'Try again', variant: 'destructive' });
   227→    } finally { setRegistering(false); }
   228→  };
   229→
   230→  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
   231→    const file = e.target.files?.[0];
   232→    if (!file) return;
   233→    setImporting(true);
   234→    setImportResult(null);
   235→    try {
   236→      const fd = new FormData();
   237→      fd.append('file', file);
   238→      const res = await fetch('/api/guesthouses/import', { method: 'POST', body: fd });
   239→      const json = await res.json();
   240→      if (!res.ok) throw new Error(json.error || 'Import failed');
   241→      setImportResult(json);
   242→      toast({ title: 'Galmeessaa Keessaa / Import Complete', description: json.message });
   243→      fetchData();
   244→    } catch (error) {
   245→      toast({ title: 'Import Failed', description: error instanceof Error ? error.message : 'Check file format', variant: 'destructive' });
   246→    } finally { setImporting(false); if (fileInputRef.current) fileInputRef.current.value = ''; }
   247→  };
   248→
   249→  const openEdit = (item: GuestHouse) => {
   250→    setEditItem(item);
   251→    setEditForm({
   252→      organizationName: item.organizationName || '',
   253→      subCity: item.subCity,
   254→      area: item.area,
   255→      specificAddress: item.specificAddress,
   256→      numberOfRooms: String(item.numberOfRooms),
   257→      licenseType: item.licenseType,
   258→      licenseLevel: item.licenseLevel,
   259→      licenseNumber: item.licenseNumber || '',
   260→      ownerName: item.ownerName || '',
   261→      contactName: item.contactName || '',
   262→      contactPhone: item.contactPhone || '',
   263→    });
   264→    setEditAreas(getAreasForSubCity(item.subCity));
   265→  };
   266→
   267→  const handleEditSubCityChange = (value: string) => {
   268→    if (!editForm) return;
   269→ setEditAreas(getAreasForSubCity(value));
   270→    setEditForm({ ...editForm, subCity: value, area: '' });
   271→  };
   272→
   273→  const handleSave = async () => {
   274→    if (!editItem || !editForm) return;
   275→    setSaving(true);
   276→    try {
   277→      const res = await fetch(`/api/guesthouses/${editItem.id}`, {
   278→        method: 'PUT',
   279→        headers: { 'Content-Type': 'application/json' },
   280→        body: JSON.stringify(editForm),
   281→      });
   282→      if (!res.ok) throw new Error('Update failed');
   283→      toast({ title: 'Updated', description: 'Record updated successfully.' });
   284→      setEditItem(null);
   285→      setEditForm(null);
   286→      fetchData();
   287→    } catch {
   288→      toast({
   289→        title: 'Update Failed',
   290→        description: 'Could not update the record.',
   291→        variant: 'destructive',
   292→      });
   293→    } finally {
   294→      setSaving(false);
   295→    }
   296→  };
   297→
   298→  const handleExportCSV = async () => {
   299→    try {
   300→      const params = new URLSearchParams();
   301→      if (filterSubCity) params.set('subCity', filterSubCity);
   302→      if (filterArea) params.set('area', filterArea);
   303→
   304→      const res = await fetch(`/api/guesthouses/export?${params}`);
   305→      if (!res.ok) throw new Error('Export failed');
   306→      const blob = await res.blob();
   307→      const url = window.URL.createObjectURL(blob);
   308→      const a = document.createElement('a');
   309→      a.href = url;
   310→      a.download = `bishoftu_guesthouses_survey_${new Date().toISOString().split('T')[0]}.csv`;
   311→      document.body.appendChild(a);
   312→      a.click();
   313→      a.remove();
   314→      window.URL.revokeObjectURL(url);
   315→      toast({ title: 'Exported', description: 'CSV file downloaded.' });
   316→    } catch {
   317→      toast({
   318→        title: 'Export Failed',
   319→        description: 'Could not export data.',
   320→        variant: 'destructive',
   321→      });
   322→    }
   323→  };
   324→
   325→  const handleExportExcel = async () => {
   326→    try {
   327→      const params = new URLSearchParams();
   328→      if (filterSubCity) params.set('subCity', filterSubCity);
   329→      if (filterArea) params.set('area', filterArea);
   330→
   331→      const res = await fetch(`/api/guesthouses/export/excel?${params}`);
   332→      if (!res.ok) throw new Error('Excel export failed');
   333→      const blob = await res.blob();
   334→      const url = window.URL.createObjectURL(blob);
   335→      const a = document.createElement('a');
   336→      a.href = url;
   337→      a.download = `bishoftu_guesthouses_survey_${new Date().toISOString().split('T')[0]}.xlsx`;
   338→      document.body.appendChild(a);
   339→      a.click();
   340→      a.remove();
   341→      window.URL.revokeObjectURL(url);
   342→      toast({ title: 'Exported', description: 'Excel file downloaded.' });
   343→    } catch {
   344→      toast({
   345→        title: 'Export Failed',
   346→        description: 'Could not export Excel file.',
   347→        variant: 'destructive',
   348→      });
   349→    }
   350→  };
   351→
   352→  return (
   353→    <div className="space-y-4">
   354→      {/* Search & Filters */}
   355→      <Card>
   356→        <CardHeader className="pb-3">
   357→          <CardTitle className="flex items-center justify-between">
   358→            <span className="flex items-center gap-2 text-lg">
   359→              <Building2 className="h-5 w-5 text-emerald-600" />
   360→              Collected Data
   361→            </span>
   362→            <Badge variant="secondary" className="text-sm">
   363→              {total} records
   364→            </Badge>
   365→          </CardTitle>
   366→          <CardDescription>View and manage all collected survey data</CardDescription>
   367→        </CardHeader>
   368→        <CardContent className="space-y-3">
   369→          <div className="relative">
   370→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   371→            <Input
   372→              placeholder="Search by name, address, owner..."
   373→              className="pl-9"
   374→              value={search}
   375→              onChange={(e) => {
   376→                setSearch(e.target.value);
   377→                setPage(1);
   378→              }}
   379→            />
   380→          </div>
   381→
   382→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   383→            <Select
   384→              value={filterSubCity}
   385→              onValueChange={handleSubCityFilterChange}
   386→            >
   387→              <SelectTrigger>
   388→                <SelectValue placeholder="Filter by sub-city" />
   389→              </SelectTrigger>
   390→              <SelectContent>
   391→                {getAllSubCities().map((sc) => (
   392→                  <SelectItem key={sc} value={sc}>{sc}</SelectItem>
   393→                ))}
   394→              </SelectContent>
   395→            </Select>
   396→
   397→            <Select
   398→              value={filterArea}
   399→              onValueChange={(v) => { setFilterArea(v); setPage(1); }}
   400→              disabled={!filterSubCity}
   401→            >
   402→              <SelectTrigger>
   403→                <SelectValue placeholder={filterSubCity ? 'Filter by area' : 'Select sub-city first'} />
   404→              </SelectTrigger>
   405→              <SelectContent>
   406→                {areas.map((a) => (
   407→                  <SelectItem key={a} value={a}>{a}</SelectItem>
   408→                ))}
   409→              </SelectContent>
   410→            </Select>
   411→          </div>
   412→
   413→          <div className="flex flex-wrap gap-2">
   414→            <Button size="sm" className="bg-blue-600 hover:bg-blue-700" onClick={() => setShowRegister(true)}>
   415→              <Plus className="mr-1 h-4 w-4" /> Registarii Haaraa
   416→            </Button>
   417→            <Button variant="outline" size="sm" onClick={() => setShowImport(true)}>
   418→              <Upload className="mr-1 h-4 w-4" /> Import Excel
   419→            </Button>
   420→            <Button variant="outline" size="sm" onClick={fetchData}>
   421→              <RefreshCw className="mr-1 h-4 w-4" /> Refresh
   422→            </Button>
   423→            <Button variant="outline" size="sm" onClick={handleExportCSV}>
   424→              <Download className="mr-1 h-4 w-4" /> CSV
   425→            </Button>
   426→            <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={handleExportExcel}>
   427→              <FileSpreadsheet className="mr-1 h-4 w-4" /> Excel
   428→            </Button>
   429→          </div>
   430→        </CardContent>
   431→      </Card>
   432→
   433→      {/* Data Table */}
   434→      {loading ? (
   435→        <Card>
   436→          <CardContent className="p-4 space-y-3">
   437→            <Skeleton className="h-10 w-full" />
   438→            {[1, 2, 3, 4].map((i) => (
   439→              <Skeleton key={i} className="h-12 w-full" />
   440→            ))}
   441→          </CardContent>
   442→        </Card>
   443→      ) : data.length === 0 ? (
   444→        <Card>
   445→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   446→            <Building2 className="mb-3 h-12 w-12 text-muted-foreground/40" />
   447→            <p className="text-lg font-medium text-muted-foreground">No records found</p>
   448→            <p className="text-sm text-muted-foreground/70">
   449→              {search || filterSubCity ? 'Try adjusting your filters' : 'Submit a survey to see data here'}
   450→            </p>
   451→          </CardContent>
   452→        </Card>
   453→      ) : (
   454→        <Card>
   455→          <CardContent className="p-0">
   456→            {/* Desktop Table */}
   457→            <div className="hidden md:block overflow-x-auto">
   458→              <table className="w-full text-sm">
   459→                <thead>
   460→                  <tr className="border-b bg-muted/50">
   461→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">#</th>
   462→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Organization / Dhaabbataa</th>
   463→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Sub-City</th>
   464→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Werreda</th>
   465→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Address</th>
   466→                    <th className="px-3 py-3 text-center font-semibold text-muted-foreground whitespace-nowrap">Rooms</th>
   467→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">License Type</th>
   468→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">License Level</th>
   469→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">License No.</th>
   470→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Owner</th>
   471→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Contact</th>
   472→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Phone</th>
   473→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Surveyor</th>
   474→                    <th className="px-3 py-3 text-left font-semibold text-muted-foreground whitespace-nowrap">Date</th>
   475→                    <th className="px-3 py-3 text-center font-semibold text-muted-foreground whitespace-nowrap">Actions</th>
   476→                  </tr>
   477→                </thead>
   478→                <tbody>
   479→                  {data.map((item, idx) => (
   480→                    <tr key={item.id} className="border-b last:border-b-0 hover:bg-muted/30 transition-colors">
   481→                      <td className="px-3 py-2.5 text-muted-foreground text-xs">{(page - 1) * 50 + idx + 1}</td>
   482→                      <td className="px-3 py-2.5 font-medium max-w-[160px] truncate" title={item.organizationName || item.guestHouseName}>
   483→                        {item.organizationName || item.guestHouseName}
   484→                      </td>
   485→                      <td className="px-3 py-2.5 whitespace-nowrap">{item.subCity}</td>
   486→                      <td className="px-3 py-2.5 whitespace-nowrap">{item.area}</td>
   487→                      <td className="px-3 py-2.5 max-w-[140px] truncate" title={item.specificAddress}>{item.specificAddress}</td>
   488→                      <td className="px-3 py-2.5 text-center">
   489→                        <span className="inline-flex items-center justify-center h-6 min-w-[24px] rounded-full bg-emerald-100 text-emerald-700 text-xs font-semibold px-1.5">
   490→                          {item.numberOfRooms}
   491→                        </span>
   492→                      </td>
   493→                      <td className="px-3 py-2.5 whitespace-nowrap"><Badge variant="outline" className="text-xs font-normal">{item.licenseType}</Badge></td>
   494→                      <td className="px-3 py-2.5 whitespace-nowrap"><Badge variant="secondary" className="text-xs font-normal">{item.licenseLevel}</Badge></td>
   495→                      <td className="px-3 py-2.5 whitespace-nowrap font-mono text-xs">{item.licenseNumber || '-'}</td>
   496→                      <td className="px-3 py-2.5 max-w-[120px] truncate" title={item.ownerName || undefined}>{item.ownerName || '-'}</td>
   497→                      <td className="px-3 py-2.5 max-w-[120px] truncate" title={item.contactName || undefined}>{item.contactName || '-'}</td>
   498→                      <td className="px-3 py-2.5 whitespace-nowrap">
   499→                        <a href={`tel:${item.contactPhone || ''}`} className="text-emerald-600 hover:underline text-xs">{item.contactPhone || '-'}</a>
   500→                      </td>
   501→                      <td className="px-3 py-2.5 max-w-[100px] truncate text-xs text-muted-foreground" title={item.surveyorName || undefined}>{item.surveyorName || '-'}</td>
   502→                      <td className="px-3 py-2.5 whitespace-nowrap text-xs text-muted-foreground">{new Date(item.createdAt).toLocaleDateString('en-GB')}</td>
   503→                      <td className="px-3 py-2.5">
